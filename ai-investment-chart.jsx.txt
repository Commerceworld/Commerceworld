function useSystemDark() {
  const [sysDark, setSysDark] = useState(
    typeof window !== "undefined" &&
      window.matchMedia &&
      window.matchMedia("(prefers-color-scheme: dark)").matches
  );
  useEffect(() => {
    if (!window.matchMedia) return;
    const mq = window.matchMedia("(prefers-color-scheme: dark)");
    const handler = (e) => setSysDark(e.matches);
    mq.addEventListener ? mq.addEventListener("change", handler) : mq.addListener(handler);
    return () => {
      mq.removeEventListener ? mq.removeEventListener("change", handler) : mq.removeListener(handler);
    };
  }, []);
  return sysDark;
}

// Props: pass darkMode={true|false} from your site's theme state to drive this
// directly (e.g. darkMode={theme === 'dark'}). If left undefined, it falls back
// to the OS/browser prefers-color-scheme setting, with a manual toggle for preview.
export default function AiInvestmentChart({ darkMode }) {
  const systemDark = useSystemDark();
  const [manualDark, setManualDark] = useState(systemDark);
  const isControlled = darkMode !== undefined;
  const dark = isControlled ? darkMode : manualDark;
  const t = dark ? themes.dark : themes.light;

  useEffect(() => {
    if (!isControlled) setManualDark(systemDark);
  }, [systemDark, isControlled]);

  function CustomDot(props) {
    const { cx, cy, payload } = props;
    if (payload.event) {
      return (
        <g>
          <circle cx={cx} cy={cy} r={7} fill={t.accent} stroke={t.bg} strokeWidth={2} />
        </g>
      );
    }
    return <circle cx={cx} cy={cy} r={4} fill={t.dotFill} stroke={t.bg} strokeWidth={1.5} />;
  }

  function CustomTooltip({ active, payload, label }) {
    if (active && payload && payload.length) {
      const d = payload[0].payload;
      return (
        <div
          style={{
            background: t.tooltipBg,
            color: t.tooltipText,
            padding: "10px 14px",
            borderRadius: 4,
            fontFamily: "'IBM Plex Mono', monospace",
            fontSize: 13,
            maxWidth: 270,
            boxShadow: "0 4px 14px rgba(0,0,0,0.3)",
          }}
        >
          <div style={{ fontWeight: 700, marginBottom: 4 }}>{label}</div>
          <div>${d.value}B global AI investment</div>
          {d.event && (
            <div style={{ marginTop: 6, color: t.tooltipSub, fontSize: 12, lineHeight: 1.4 }}>
              ▲ {d.event}
            </div>
          )}
        </div>
      );
    }
    return null;
  }

  return (
    <div
      style={{
        fontFamily: "'IBM Plex Mono', 'Courier New', monospace",
        background: t.bg,
        padding: "32px 24px",
        maxWidth: 880,
        margin: "0 auto",
        color: t.text,
        transition: "background 0.2s ease, color 0.2s ease",
        position: "relative",
      }}
    >
      {!isControlled && (
        <button
          onClick={() => setManualDark(!manualDark)}
          style={{
            position: "absolute",
            top: 20,
            right: 20,
            background: "transparent",
            border: `1px solid ${t.subtext}`,
            color: t.subtext,
            fontFamily: "'IBM Plex Mono', monospace",
            fontSize: 11,
            padding: "4px 10px",
            borderRadius: 3,
            cursor: "pointer",
          }}
        >
          {dark ? "☀ Light" : "☾ Dark"}
        </button>
      )}

      <div style={{ marginBottom: 4, fontSize: 12, letterSpacing: 2, color: t.eyebrow, fontWeight: 700 }}>
        ECONOMICPULSE · AI ECONOMY
      </div>
      <h2
        style={{
          fontFamily: "Georgia, 'Times New Roman', serif",
          fontSize: 28,
          margin: "0 0 4px 0",
          color: t.text,
        }}
      >
        Global AI Investment, 2018–2026
      </h2>
      <p style={{ fontSize: 13, color: t.subtext, margin: "0 0 24px 0", maxWidth: 640 }}>
        Estimated global private AI investment (US$ billions). The boom accelerated sharply
        after ChatGPT's 2022 launch, as hyperscalers and sovereign funds poured capital into
        AI infrastructure.
      </p>

      <ResponsiveContainer width="100%" height={360}>
        <LineChart data={data} margin={{ top: 20, right: 20, left: 0, bottom: 5 }}>
          <CartesianGrid stroke={t.grid} strokeDasharray="0" vertical={false} />
          <XAxis
            dataKey="year"
            stroke={t.axis}
            tick={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 12, fill: t.axis }}
            axisLine={{ stroke: t.axis }}
            tickLine={false}
          />
          <YAxis
            stroke={t.axis}
            tick={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 12, fill: t.axis }}
            axisLine={false}
            tickLine={false}
            tickFormatter={(v) => `$${v}B`}
          />
          <Tooltip content={<CustomTooltip />} />
          <Line
            type="monotone"
            dataKey="value"
            stroke={t.line}
            strokeWidth={2.5}
            dot={<CustomDot />}
            activeDot={{ r: 8 }}
          />
        </LineChart>
      </ResponsiveContainer>

      <div style={{ marginTop: 28, borderTop: `2px solid ${t.divider}`, paddingTop: 16 }}>
        <div style={{ fontSize: 12, letterSpacing: 1.5, color: t.eyebrow, fontWeight: 700, marginBottom: 10 }}>
          WHY INVESTMENT JUMPED
        </div>
        {hikes.map((h) => (
          <div key={h.year} style={{ display: "flex", gap: 14, marginBottom: 12 }}>
            <div
              style={{
                fontFamily: "Georgia, serif",
                fontWeight: 700,
                color: t.text,
                fontSize: 15,
                minWidth: 46,
              }}
            >
              {h.year}
            </div>
            <div style={{ fontSize: 13.5, lineHeight: 1.5, color: t.text }}>
              <span style={{ color: t.accent, fontWeight: 700 }}>${h.value}B — </span>
              {h.event}
            </div>
          </div>
        ))}
      </div>

      <div style={{ marginTop: 20, fontSize: 11, color: t.faint }}>
        Figures are approximate global private AI investment estimates compiled from Stanford
        HAI AI Index and industry funding trackers; methodology varies by source, so treat as
        directional rather than exact. 2026 is a year-to-date estimate.
      </div>
    </div>
  );
}
